cmake_minimum_required(VERSION 3.20)
project(rapid_group_experiment LANGUAGES CXX)
find_package(Threads REQUIRED)
enable_testing()

add_library(rapid_group_runtime experiment.cpp)
target_compile_features(rapid_group_runtime PUBLIC cxx_std_20)
target_include_directories(rapid_group_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
target_link_libraries(rapid_group_runtime PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(rapid_group_runtime PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(rapid_group_probe probe.cpp)
target_link_libraries(rapid_group_probe PRIVATE rapid_group_runtime)
add_library(rapid_group_diagnostic_runtime experiment.cpp)
target_compile_features(rapid_group_diagnostic_runtime PUBLIC cxx_std_20)
target_include_directories(rapid_group_diagnostic_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
target_compile_definitions(rapid_group_diagnostic_runtime PUBLIC OOX_GROUP_DIAGNOSTICS=1)
target_link_libraries(rapid_group_diagnostic_runtime PUBLIC Threads::Threads)
add_executable(rapid_group_diagnose probe.cpp)
target_link_libraries(rapid_group_diagnose PRIVATE rapid_group_diagnostic_runtime)
add_executable(rapid_group_tests tests.cpp)
target_link_libraries(rapid_group_tests PRIVATE rapid_group_diagnostic_runtime)
add_test(NAME rapid_group_oracles COMMAND rapid_group_tests --workers 4)
set_tests_properties(rapid_group_oracles PROPERTIES TIMEOUT 180)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME rapid_group_runner COMMAND ${Python3_EXECUTABLE} -B
    "${CMAKE_CURRENT_SOURCE_DIR}/test_runner.py" $<TARGET_FILE:rapid_group_probe>)
  set_tests_properties(rapid_group_runner PROPERTIES TIMEOUT 60)
endif()
